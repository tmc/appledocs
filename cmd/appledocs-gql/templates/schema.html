GraphQL Schema for Apple Documentation

This schema provides a comprehensive model of Apple's documentation structure, designed for querying through the GraphQL API.

Core Types

Document

Represents an Apple documentation item (framework, class, method, property, or other documentation element).

type Document {
  id: ID!                        # Unique identifier for the document
  path: String!                  # Path to the document in the documentation hierarchy
  title: String                  # Document title
  abstract: String               # Brief description of the document
  framework: String              # Framework the document belongs to
  type: String                   # Document type (class, method, protocol, etc.)
  kind: String                   # Document kind (symbol, article, collection, etc.)
  role: String                   # Role of the document (collection, symbol, article, etc.)
  primaryContentSections: [ContentSection]  # Content sections of the document
  topicSections: [TopicSection]  # Topic sections grouping related content
  platforms: [Platform]          # Available platforms for this API (iOS, macOS, etc.)
  isDeprecated: Boolean          # Whether this API is deprecated
  modules: [String]              # Module names associated with this document
  interfaceLanguages: [String]   # Language variants available (swift, occ, data)
  schemaVersion: SchemaVersion   # Schema version of the document
  hierarchy: Hierarchy           # Hierarchy information for navigation
  metadata: JSON                 # Document metadata
  content: JSON                  # Full document content
  references: [Reference]        # Related documents
  parent: Document               # Parent document
  children: [Document]           # Child documents
  sections: [Section]            # Original document sections (for backward compatibility)
}

Platform

Platform availability information for APIs.

type Platform {
  name: String!            # Platform name (e.g., iOS, macOS)
  introducedAt: String     # Introduction version (e.g., 13.0)
  deprecatedAt: String     # Deprecated version (if applicable)
  beta: Boolean            # Beta status
}

Technology

Technology/Framework information (from technologies.json).

type Technology {
  id: ID!                  # Technology identifier
  title: String!           # Technology title
  path: String!            # Technology path
  abstract: String         # Brief description
  languages: [String]      # Supported languages
  tags: [String]           # Tags for categorization
  document: Document       # The document containing detailed information
}

Content Structure Types

ContentSection

type ContentSection {
  title: String                # Section title
  anchor: String               # Section anchor for linking
  kind: String                 # Section type (declarations, discussion, parameters, etc.)
  content: [ContentItem]       # Section content items
}

TopicSection

type TopicSection {
  title: String                # Section title
  anchor: String               # Section anchor for linking
  identifiers: [String]        # Identifiers of documents in this section
  topics: [Document]           # Documents in this section
}

ContentItem

interface ContentItem {
  type: String                 # Content type (text, code, image, etc.)
}

type TextContent implements ContentItem {
  type: String                 # Content type
  text: String                 # Text content
}

type CodeContent implements ContentItem {
  type: String                 # Content type
  code: String                 # Code text
  language: String             # Programming language
}

type ReferenceContent implements ContentItem {
  type: String                 # Content type
  identifier: String           # Reference identifier
  reference: Reference         # Referenced document
}

Reference

type Reference {
  identifier: String!          # Reference identifier
  title: String                # Reference title
  role: String                 # Reference role
  type: String                 # Reference type
  url: String                  # Reference URL
  isActive: Boolean            # Whether the reference is active
  abstract: String             # Brief description
  document: Document           # The referenced document
}

API Index Types

APIIndex

type APIIndex {
  interfaceLanguage: String!   # Interface language (swift, occ, data)
  modules: [ModuleIndex]       # Top-level modules
}

ModuleIndex & Module Children

type ModuleIndex {
  path: String!                # Module path
  title: String!               # Module title
  type: String                 # Module type
  children: [ModuleChild]      # Child elements of the module
}

interface ModuleChild {
  path: String                 # Element path
  title: String                # Element title
  type: String                 # Element type
  deprecated: Boolean          # Whether the element is deprecated
}

Index Element Types

type ClassIndex implements ModuleChild {
  path: String                 # Class path
  title: String                # Class title
  type: String                 # Element type (class)
  deprecated: Boolean          # Whether the class is deprecated
  children: [ModuleChild]      # Child elements of the class
}

type MethodIndex implements ModuleChild {
  path: String                 # Method path
  title: String                # Method signature
  type: String                 # Element type (method)
  deprecated: Boolean          # Whether the method is deprecated
}

type PropertyIndex implements ModuleChild {
  path: String                 # Property path
  title: String                # Property name
  type: String                 # Element type (property)
  deprecated: Boolean          # Whether the property is deprecated
}

type SectionMarker implements ModuleChild {
  path: String                 # Path (usually empty)
  title: String                # Section title
  type: String                 # Element type (groupMarker)
  deprecated: Boolean          # Always false for group markers
}

Queries

type Query {
  document(path: String!, language: String): Document
  # Get a document by its path
  
  search(query: String!, limit: Int = 10, language: String): SearchResult!
  # Search for documents matching a query
  
  technologies: [TechnologyGroup!]!
  # Get the technologies index (top-level frameworks organized by group)
  
  frameworks: [Framework!]!
  # List all top-level frameworks
  
  framework(id: String!, language: String): Framework
  # Get a specific framework by its identifier
  
  apiIndex(framework: String!, language: String = "swift"): APIIndex
  # Get an API index for a specific framework
  
  documentsByType(type: String!, framework: String, language: String): [Document!]!
  # List documents of a specific type
  
  documentsByPlatform(platform: String!, framework: String, language: String): [Document!]!
  # List documents available on a specific platform
  
  relatedDocuments(path: String!, limit: Int = 5, language: String): [Document!]!
  # Get related documents for a specific document
  
  platforms: [Platform!]!
  # Get platform information
}
